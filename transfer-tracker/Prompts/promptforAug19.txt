RECENT ACTIVITY SUMMARY
==================================================

There is a function:

function updateRecentActivitySummary(transfer) {

    const summary =
        document.getElementById(
            "activitySubtitle"
        );

    if (!summary) {
        return;
    }

    if (!transfer) {

        summary.textContent =
            "No transfers yet";

        return;
    }

    const status =
        transfer.status === "approved"
            ? "Approved"
            : "Declined";

    const time =
        new Date(
            transfer.timestamp
        ).toLocaleTimeString(
            [],
            {
                hour: "numeric",
                minute: "2-digit"
            }
        );

    summary.textContent =
        `Last transfer · ${status} · ${time}`;

}

This should remain functional.
