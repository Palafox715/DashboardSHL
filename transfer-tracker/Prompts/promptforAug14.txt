Critical Threshold Behavior

Transfers are always stored in IndexedDB.

Being below a threshold does not delete or exclude transfers.

Example:

2 approved

Both transfers remain visible.

Bonus:

Approved Bonus: $0.00

When the third arrives:

Approved Bonus: $14.10

If one is deleted:

2 approved
Approved Bonus: $0.00

The category is recalculated from the remaining raw transfers.

Never subtract an individual transfer's old rate.

Import Duplicate Protection

The final importTransfers() implementation was changed so it checks existing IndexedDB IDs before adding records.

This avoids IndexedDB ConstraintError problems.

Current behavior is:

Existing IDs are skipped.
New IDs are imported.
Nothing is overwritten.

Important Backup Rule

The user originally wondered whether overwriting was really a concern because importing would generally happen on a new machine.

We agreed the safest user-facing behavior is:

Never overwrite an existing transfer. Skip duplicate IDs and report how many were skipped.

This is now implemented.

Current Backup Code

The important storage.js function is now:

function importTransfers(transfers) {

    return new Promise((resolve, reject) => {

        const readTransaction =
            db.transaction(
                STORE_NAME,
                "readonly"
            );


        const readStore =
            readTransaction.objectStore(
                STORE_NAME
            );


        const request =
            readStore.getAllKeys();


        request.onsuccess = function() {

            const existingIds =
                new Set(request.result);


            const newTransfers =
                transfers.filter(
                    transfer =>
                        !existingIds.has(
                            transfer.id
                        )
                );


            const skippedCount =
                transfers.length -
                newTransfers.length;


            if (!newTransfers.length) {

                resolve({
                    importedCount: 0,
                    skippedCount
                });

                return;

            }


            const transaction =
                db.transaction(
                    STORE_NAME,
                    "readwrite"
                );


            const store =
                transaction.objectStore(
                    STORE_NAME
                );


            let importedCount = 0;


            newTransfers.forEach(
                transfer => {

                    const addRequest =
                        store.add(transfer);


                    addRequest.onsuccess =
                        function() {

                            importedCount++;

                        };

                }
            );


            transaction.oncomplete =
                function() {

                    resolve({

                        importedCount,

                        skippedCount

                    });

                };


            transaction.onerror =
                function() {

                    reject(
                        transaction.error ||
                        new Error(
                            "Import transaction failed."
                        )
                    );

                };


            transaction.onabort =
                function() {

                    reject(
                        transaction.error ||
                        new Error(
                            "Import transaction was aborted."
                        )
                    );

                };

        };


        request.onerror = function() {

            reject(
                request.error ||
                new Error(
                    "Unable to check existing transfers."
                )
            );

        };

    });

}

Do not modify this without a specific reason.
